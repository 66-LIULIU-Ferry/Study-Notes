[小红书面试题] 3 个线程，按顺序交替输出数字1-20

```java
import java.util.Scanner;

public class Main {
    private static int current = 1;     // 目前要打印的数字
    private static int threadTurn = 0;  // 现在轮到的线程
    private static final Object lock = new Object();

    public static void main(String[] args) throws InterruptedException {
        Thread[] threads = new Thread[3];
        for (int i = 0; i < 3; i++) {
            final int threadId = i;
            threads[i] = new Thread(() -> {
                while (true) {
                    synchronized (lock) {
                        while (threadTurn != threadId && current <= 20) {
                            try {
                                lock.wait();
                            } catch (InterruptedException e) {
                                return;
                            }
                        }
                        if (current > 20) {
                            lock.notifyAll();
                            return;
                        }
                        System.out.println("Thread " + threadId + ": " + current);
                        current++;
                        threadTurn = (threadTurn + 1) % 3;
                        lock.notifyAll();
                    }
                }
            });
            threads[i].start();
        }

        // 等所有线程跑完
        for (Thread t : threads) {
            t.join();
        }
    }
}
```
